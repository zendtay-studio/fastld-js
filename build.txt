# Download the reference corpora used to build the language database.
# This script clones two repositories and moves their content into the local
# languages folder so the build step can generate data.json.

# Clone the language corpus repository.
git clone https://github.com/patrickschur/language-detection.git

# Clone the frequency word repository.
git clone https://github.com/hermitdave/FrequencyWords.git

# Move the downloaded corpus folders into the expected local structure and build the index.
mv language-detection/resources/* languages && mv FrequencyWords/content/2018/* languages && node build.js
